riscv64-unknown-elf-gcc -march=rv32i -mabi=ilp32 -g -ffreestanding -O0 -nostdlib -nostartfiles -c ejemplo.c -o ejemplo.o

	start_juan.s ejemplo.o \
	-o ejemplo.out

	ejemplo.out > ejemplo.lst

	ejemplo.out ejemplo.bin
